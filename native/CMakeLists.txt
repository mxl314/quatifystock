cmake_minimum_required(VERSION 3.18)
project(quant_gateway_bridges LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(ESUNNY_TRADE_SDK "" CACHE PATH "Directory containing DstarTradeApi.h and the lib directory")
set(ESUNNY_QUOTE_SDK "" CACHE PATH "Directory containing DstarQuoteApi.h and the lib directory")
set(CTP_MD_SDK "" CACHE PATH "Directory containing ThostFtdcMdApi.h and thostmduserapi_se.lib")
set(CTP_TD_SDK "" CACHE PATH "Directory containing ThostFtdcTraderApi.h and thosttraderapi_se.lib")

if(ESUNNY_TRADE_SDK AND ESUNNY_QUOTE_SDK)
  add_library(esunny_v10_bridge SHARED esunny_v10_bridge.cpp)
  target_include_directories(esunny_v10_bridge PRIVATE "${ESUNNY_TRADE_SDK}/include")
  add_library(esunny_v10_quote_bridge SHARED esunny_v10_quote_bridge.cpp)
  target_include_directories(esunny_v10_quote_bridge PRIVATE "${ESUNNY_QUOTE_SDK}/include")
  if(WIN32)
    target_link_libraries(esunny_v10_bridge PRIVATE "${ESUNNY_TRADE_SDK}/lib/win/x64/libdstartradeapi.lib")
    target_link_libraries(esunny_v10_quote_bridge PRIVATE "${ESUNNY_QUOTE_SDK}/lib/win/x64/libdstarquoteapi.lib")
  else()
    target_link_libraries(esunny_v10_bridge PRIVATE "${ESUNNY_TRADE_SDK}/lib/linux/libdstartradeapi.so")
    target_link_libraries(esunny_v10_quote_bridge PRIVATE "${ESUNNY_QUOTE_SDK}/lib/linux/libdstarquoteapi.so")
  endif()
endif()

if(CTP_MD_SDK)
  add_library(ctp_md_bridge SHARED ctp/ctp_md_bridge.cpp)
  target_include_directories(ctp_md_bridge PRIVATE "${CTP_MD_SDK}")
  target_link_libraries(ctp_md_bridge PRIVATE "${CTP_MD_SDK}/thostmduserapi_se.lib")
endif()

if(CTP_TD_SDK)
  add_library(ctp_td_bridge SHARED ctp/ctp_td_bridge.cpp)
  target_include_directories(ctp_td_bridge PRIVATE "${CTP_TD_SDK}")
  target_link_libraries(ctp_td_bridge PRIVATE "${CTP_TD_SDK}/thosttraderapi_se.lib")
endif()
